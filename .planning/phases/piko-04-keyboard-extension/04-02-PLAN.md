<objective>
Implement the stablePrefix-based streaming text insertion that makes dictation feel responsive
rather than laggy. As partial transcripts arrive via `CaptureDraft`, text appears in the host
field immediately — but only the unstable tail (characters beyond `stablePrefix`) gets rewritten
on each update, preventing the jarring visual thrash where early words jump around.

Purpose: CAPT-01 ("partial transcript streams into the host text field") and CAPT-02 ("only
rewrites characters after stablePrefix") are the phase's core delivery. This is what makes Piko
feel like "thinking out loud" rather than "wait for transcription." The implementation must be
correct before Phase 5 plugs in real transcription — incorrect insertion logic would either
cause text thrash (rewriting too much) or text loss (rewriting too little).

Output: `TextInsertionController` class encapsulating the diffing/insertion logic, wired into
`KeyboardViewController.applyDraft()`, with comprehensive tests using synthetic `CaptureDraft`
sequences that prove correctness without requiring a real transcriber.
</objective>

public struct CaptureDraft: Codable, Sendable, Equatable {
    public var sessionEpoch: Int
    public var sequence: Int
    public var text: String
    public var stablePrefix: Int
    public var startedAt: Date

    public func isNewer(than previous: CaptureDraft?) -> Bool {
        guard let previous else { return true }
        if sessionEpoch != previous.sessionEpoch { return sessionEpoch > previous.sessionEpoch }
        return sequence > previous.sequence
    }
}

public struct CaptureResult: Codable, Sendable, Equatable {
    public var id: UUID
    public var raw: String
    public var shipped: String
    public var route: Route
    public var profile: Profile
    public var createdAt: Date
    public var timings: Timings
}

From 04-RESEARCH.md Pattern 1 (TextInsertionController):
```swift
final class TextInsertionController {
    private weak var proxy: UITextDocumentProxy?
    private var insertedChars: Int = 0
    private var lastApplied: CaptureDraft?

    init(proxy: UITextDocumentProxy) {
        self.proxy = proxy
    }

    func apply(_ draft: CaptureDraft) {
        guard let proxy, draft.isNewer(than: lastApplied) else { return }

        // How many stable chars did we already insert?
        let alreadyStable = lastApplied.map { min($0.stablePrefix, insertedChars) } ?? 0

        // Delete only the unstable tail we previously inserted
        let toDelete = insertedChars - alreadyStable
        for _ in 0..<toDelete { proxy.deleteBackward() }

        // Insert from where stable portion ends
        let newText = String(draft.text.dropFirst(alreadyStable))
        proxy.insertText(newText)

        insertedChars = draft.text.count
        lastApplied = draft
    }

    func commit(_ result: CaptureResult) {
        guard let proxy else { return }

        // Delete everything we streamed
        for _ in 0..<insertedChars { proxy.deleteBackward() }

        // Insert the final shipped text
        proxy.insertText(result.shipped)

        // Reset for next session
        insertedChars = 0
        lastApplied = nil
    }

    func reset() {
        insertedChars = 0
        lastApplied = nil
    }
}
```

From 04-RESEARCH.md Anti-Pattern 1 (Full Replacement — what NOT to do):
```swift
// BAD: User sees "Hello" jump to "Hello world" to "Hello world how" — the entire string
// gets deleted and reinserted on every update, causing visible flicker.
func applyDraft() {
    for _ in 0..<lastLength { proxy.deleteBackward() }  // Delete everything
    proxy.insertText(draft.text)                         // Reinsert everything
    lastLength = draft.text.count
}
```
