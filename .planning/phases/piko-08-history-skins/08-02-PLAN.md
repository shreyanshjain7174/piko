import PikoKit

public actor SQLiteMemory: Memory {
    public init(path: URL) { /* TODO */ }

    public func record(_ result: CaptureResult) async {}

    public func lexicon(limit: Int) async -> [String] { [] }
    public func nearestEdits(to text: String, limit: Int) async -> [EditPair] { [] }
    public func search(_ query: String, limit: Int) async -> [CaptureResult] { [] }
}

public actor EphemeralMemory: Memory {
    private var results: [CaptureResult] = []
    private var edits: [EditPair] = []
    public init() {}

    public func record(_ result: CaptureResult) { results.append(result) }
    public func recordEdit(_ pair: EditPair) { edits.append(pair) }
    public func lexicon(limit: Int) -> [String] { [] }
    public func nearestEdits(to text: String, limit: Int) -> [EditPair] { Array(edits.suffix(limit)) }
    public func search(_ query: String, limit: Int) -> [CaptureResult] {
        results.filter { $0.shipped.localizedCaseInsensitiveContains(query) }.suffix(limit).reversed()
    }
}

Verified FTS5 schema/query forms (sqlite.org/fts5.html, fetched by research this session):
```sql
CREATE TABLE IF NOT EXISTS results (
    id TEXT NOT NULL UNIQUE, raw TEXT NOT NULL, shipped TEXT NOT NULL,
    route TEXT NOT NULL, profile TEXT NOT NULL, createdAt REAL NOT NULL,
    firstWordMS INTEGER NOT NULL, transcribeMS INTEGER NOT NULL, brainMS INTEGER NOT NULL
);
CREATE VIRTUAL TABLE IF NOT EXISTS results_fts USING fts5(
    raw, shipped, content='results', content_rowid='rowid'
);
CREATE TRIGGER IF NOT EXISTS results_ai AFTER INSERT ON results BEGIN
    INSERT INTO results_fts(rowid, raw, shipped) VALUES (new.rowid, new.raw, new.shipped);
END;
CREATE TABLE IF NOT EXISTS edits (
    raw TEXT NOT NULL, shipped TEXT NOT NULL, final TEXT NOT NULL,
    profile TEXT NOT NULL, createdAt REAL NOT NULL
);
```
Only `AFTER INSERT` is built — this phase never updates or deletes a history row (no UI feature
does either), and SQLite's own docs warn an external-content table with incomplete trigger
coverage silently drifts out of sync the moment an omitted operation occurs, so `AFTER UPDATE`/
`AFTER DELETE` must be added together, not incrementally, only if a future phase adds history
mutation. Do not add them speculatively now.

FTS5 MATCH query, joined back to `results` for the non-indexed columns (verified pattern):
```sql
SELECT r.id, r.raw, r.shipped, r.route, r.profile, r.createdAt,
       r.firstWordMS, r.transcribeMS, r.brainMS
FROM results_fts f JOIN results r ON r.rowid = f.rowid
WHERE results_fts MATCH ? ORDER BY rank LIMIT ?;
```
`ORDER BY rank` uses FTS5's built-in bm25-backed `rank` column — faster than calling `bm25()`
directly, per research's citation.

**Empty-query design (a deliberate refinement over research's own illustrative code sample, not a
protocol change):** an empty/whitespace-only query string must never be handed to `MATCH` — FTS5
treats an empty match expression as invalid syntax, not "match everything." `EphemeralMemory`'s own
`search("")` effectively returns every result (`localizedCaseInsensitiveContains("")` is always
true) most-recent-first via `.suffix(limit).reversed()`. To keep that same "every session is
visible with no query typed" behavior for `HIST-01` while still using real FTS5 for actual search
text (`HIST-02`), branch inside `search(_:limit:)`: a trimmed-empty query runs
`SELECT ... FROM results ORDER BY createdAt DESC LIMIT ?` directly against the content table
(bypassing `MATCH` entirely); a non-empty query runs the sanitized `MATCH`+`rank` query above. This
is the one place this plan's behavior is not a byte-for-byte copy of `EphemeralMemory`'s code, but
it reproduces the same observable "no query = show every recorded session, newest first" result.

**Substring-vs-token search is the other honest divergence to build, not paper over:**
`EphemeralMemory.search` does a true substring scan (`localizedCaseInsensitiveContains`), so
searching "runn" finds a result containing "running". FTS5's default tokenizer matches whole
tokens only — searching "runn" will **not** find "running" unless the query uses FTS5's `*` prefix
operator, which `HIST-02` does not ask for. Tests must assert the real FTS5 token-matching
behavior (a partial mid-word substring does NOT match), not force substring semantics that would
require a different (trigram) tokenizer never researched or requested — see Pitfall 2/Common
Pitfalls in `08-RESEARCH.md`.

FTS5 query sanitizer (verified pattern, Pitfall 2 — build this exactly):
```swift
static func sanitizedMatchExpression(for query: String) -> String {
    let tokens = query.split(separator: " ").map { token -> String in
        "\"\(token.replacingOccurrences(of: "\"", with: "\"\""))\""
    }
    return tokens.joined(separator: " AND ")
}
```
This wraps every whitespace-separated token in double quotes (FTS5 treats a quoted string as a
literal even if the token itself is a reserved word like `AND`/`NOT`/`NEAR`) and joins tokens with
an explicit `AND` — `HIST-02` asks for "searchable by content," not boolean/phrase search
operators, so this is the correct, complete scope, not a partial implementation.
