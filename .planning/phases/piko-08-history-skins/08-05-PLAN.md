@main

    @State private var armError: String?
    @State private var hasArmedThisLaunch = false
    @State private var staleAtLaunch = false

    private var showReArmBanner: Bool {
        (hasArmedThisLaunch && phaseText == "idle") || staleAtLaunch
    }

    var body: some View {
        // TODO: history list, skin picker.
        VStack(spacing: 16) {
            Text("Piko")
            Text("Session: \(phaseText)")
            if phaseText == SessionPhase.capturing.rawValue {
                Text("Recording...")
            } else if phaseText == SessionPhase.tidying.rawValue {
                Text("Processing...")
            }
            if showReArmBanner {
                Text("Session ended — tap Arm to re-arm")
            }
            Button("Arm") {
                Task {
                    do {
                        try await AppComposition.shared.armSession()
                        hasArmedThisLaunch = true
                        staleAtLaunch = false
                        armError = nil
                    } catch {
                        armError = "\(error)"
                    }
                }
            }
            if let armError {
                Text(armError)
            }
        }
        .task {
            if let state = AppComposition.shared.channel.readState(), !state.isLive() {
                staleAtLaunch = true
            }
        }
        .task {
            for await phase in AppComposition.shared.session.phase {
                phaseText = phase.rawValue
            }
        }
    }
}

`SessionState`/`SessionChannel`/`Skin` (verbatim, `Sources/PikoKit/Contracts.swift`/`Protocols.swift`
— unchanged by this plan):
```swift
public struct SessionState: Codable, Sendable, Equatable {
    public var phase: SessionPhase
    public var heartbeat: Date
    public var skin: Skin
    public var profile: Profile
    public init(phase: SessionPhase = .idle, heartbeat: Date = .now, skin: Skin = .cute, profile: Profile = .message) { ... }
}
public enum Skin: String, Codable, Sendable, CaseIterable, Hashable { case cute, cool, hero, sparkle }
public protocol SessionChannel: Sendable {
    func readState() -> SessionState?
    func writeState(_ state: SessionState)
    ...
}
```

**Skin picker reuses the existing read-modify-write mechanism** (`08-RESEARCH.md` Pattern 6,
mirroring the existing `existing?.skin ?? .cute` read pattern already used elsewhere in this
codebase for `profile`) — this is what `App/Piko/SkinSelection.swift` (new file, this plan) makes
independently testable:
```swift
enum SkinSelection {
    static func apply(_ skin: Skin, via channel: any SessionChannel) {
        var state = channel.readState() ?? SessionState()
        state.skin = skin
        channel.writeState(state)
    }
}
```
This file lives in `App/Piko/` and is **not** in `PikoCaptureCore`'s `exclude:` list (only
`AppComposition.swift`/`PikoApp.swift`/`PikoShortcuts.swift`/`ArmSessionIntent.swift`/`Info.plist`/
`Piko.entitlements` are excluded) — it is automatically included in both the Xcode `Piko` app
target (`sources: [Piko]`) and the SPM `PikoCaptureCore` test-visible target, with zero
`project.yml`/`Package.swift` change needed. `MockSessionChannel` is already defined as a
same-target-visible type inside `Tests/PikoTranscribeTests/CaptureIntegrationTests.swift` (a
`final class` at file scope, visible to any other file in the same `PikoTranscribeTests` module) —
`SkinSelectionTests.swift` can construct `MockSessionChannel()` directly with no new test double.

**History list data flow:** `AppComposition.shared.memory.search(searchText, limit: 50)` is `async`
— drive it from a `.task(id: searchText)` modifier (re-runs whenever `searchText` changes,
including the initial empty value at view appearance, which correctly loads "every session" per
Plan 08-02's empty-query design) assigning into a local `@State private var results:
[CaptureResult] = []`. `CaptureResult` has `id: UUID`, `shipped: String` (what was inserted — show
this as the primary line), `raw: String` (what was heard — a reasonable secondary/subtitle line for
context, not required but genuinely useful and cheap to add), `createdAt: Date`, `profile: Profile`.

**PikoFace's skin input:** there is no dedicated async stream for skin changes; the existing `.task`
that already loops `for await phase in AppComposition.shared.session.phase` fires on every phase
transition — read `AppComposition.shared.channel.readState()?.skin ?? .cute` into a `@State private
var currentSkin: Skin = .cute` inside that same loop body (one extra line, no new observation
mechanism invented) so `PikoFace`'s skin updates land in step with phase changes and after the
picker itself changes it via `SkinSelection.apply` + a local `@State` update in the picker's own
`onChange` (both together keep the displayed skin correct immediately on selection and correct
after any phase tick).
