Create `CaptureCoordinator` that owns the transcription lifecycle:

```swift
import Foundation
import PikoKit
import PikoBridge
import PikoAudio
import PikoTranscribe
import AVFAudio

/// Orchestrates capture: buffers → transcriber → drafts → channel.
/// Lives in the container app only; the keyboard is just a remote control.
@MainActor
public final class CaptureCoordinator {
    private let session: SessionCoordinator
    private let channel: any SessionChannel
    private let transcriber: any Transcriber

    private var transcriptionTask: Task<Void, Never>?
    private var sessionEpoch = 0

    public init(session: SessionCoordinator,
                channel: any SessionChannel,
                transcriber: any Transcriber) {
        self.session = session
        self.channel = channel
        self.transcriber = transcriber
    }

    /// Start capture: begin audio recording and transcription pipeline.
    public func startCapture() async throws {
        sessionEpoch += 1
        try await session.startCapture()

        // Configure transcriber with buffer stream
        if let engine = transcriber as? SpeechTranscriberEngine {
            await engine.configure(buffers: session.buffers, sessionEpoch: sessionEpoch)
        }

        // Start consuming hypotheses and forwarding to channel
        transcriptionTask = Task { [weak self] in
            guard let self else { return }
            for await draft in self.transcriber.hypotheses() {
                self.channel.writeDraft(draft)
                self.channel.post(.draftUpdated)
            }
        }
    }

    /// Stop capture: finalize transcription and write result.
    public func stopCapture() async {
        await session.stopCapture()
        transcriptionTask?.cancel()
        transcriptionTask = nil

        // Get final transcript
        let finalText = await transcriber.finish()

        // Write result
        let result = CaptureResult(
            raw: finalText,
            shipped: finalText,  // TODO(phase 6): Brain rewrites this
            route: .write
        )
        channel.writeResult(result)
        channel.post(.resultReady)
    }

    /// Respond to keyboard signals
    public func handleSignal(_ signal: Signal) async {
        switch signal {
        case .captureStart:
            do {
                try await startCapture()
            } catch {
                // Log error; keyboard will see phase didn't change
                print("CaptureCoordinator: startCapture failed: \(error)")
            }
        case .captureStop:
            await stopCapture()
        default:
            break
        }
    }
}
```

**Key points:**
- `CaptureCoordinator` sits between `SessionCoordinator` (audio) and `Transcriber` (speech-to-text)
- On `startCapture()`: tell session to start, configure transcriber with buffer stream, spawn task to forward hypotheses → channel
- On `stopCapture()`: tell session to stop, cancel transcription task, get final text, write result
- The `handleSignal()` method will be wired from AppComposition to respond to keyboard's `.captureStart`/`.captureStop`

```swift
import Testing
@testable import PikoKit
@testable import PikoTranscribe
#if os(iOS)
@testable import PikoAudio

/// Integration tests for the capture pipeline using mocks
@Suite struct CaptureIntegrationTests {

    #if os(iOS)
    @Test @MainActor func startCaptureBeginsDraftFlow() async throws {
        // Setup
        let channel = MockSessionChannel()
        let mock = MockTranscriber()
        let script = MockTranscriber.Script(drafts: [
            CaptureDraft(sessionEpoch: 1, sequence: 1, text: "Hello", stablePrefix: 0),
            CaptureDraft(sessionEpoch: 1, sequence: 2, text: "Hello world", stablePrefix: 11),
        ], delayBetween: .milliseconds(50))
        await mock.setScript(script)

        // Create session with mocks
        let session = SessionCoordinator(
            channel: channel,
            interruptions: NullInterruptionSource(),
            isForeground: { true }
        )

        let coordinator = CaptureCoordinator(
            session: session,
            channel: channel,
            transcriber: mock
        )

        // Start
        try await session.arm()
        try await coordinator.startCapture()

        // Wait for drafts to arrive
        try await Task.sleep(for: .milliseconds(200))

        // Verify
        let drafts = channel.allDrafts()
        #expect(drafts.count >= 2)
        #expect(drafts.last?.text == "Hello world")

        // Cleanup
        await coordinator.stopCapture()
        await session.disarm()
    }

    @Test @MainActor func stopCaptureWritesResult() async throws {
        // Setup
        let channel = MockSessionChannel()
        let mock = MockTranscriber()
        let script = MockTranscriber.Script(drafts: [
            CaptureDraft(sessionEpoch: 1, sequence: 1, text: "Final text.", stablePrefix: 11),
        ], delayBetween: .milliseconds(10))
        await mock.setScript(script)

        let session = SessionCoordinator(
            channel: channel,
            interruptions: NullInterruptionSource(),
            isForeground: { true }
        )

        let coordinator = CaptureCoordinator(
            session: session,
            channel: channel,
            transcriber: mock
        )

        // Capture flow
        try await session.arm()
        try await coordinator.startCapture()
        try await Task.sleep(for: .milliseconds(100))
        await coordinator.stopCapture()

        // Verify result
        let result = channel.readResult()
        #expect(result != nil)
        #expect(result?.raw == "Final text.")

        await session.disarm()
    }
    #endif
}
```
